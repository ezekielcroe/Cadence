        do {
            // Configure SwiftData model container
            let schema = Schema([
                Activity.self,
                ActivitySet.self,
                TimeBlock.self,
                TimeBlockRule.self
            ])
            
            let modelConfiguration = ModelConfiguration(
                schema: schema,
                isStoredInMemoryOnly: false
            )
            
            modelContainer = try ModelContainer(
                for: schema,
                configurations: [modelConfiguration]
            )
            
            // Create default activity sets if needed
            setupDefaultData()

            requestNotificationPermissions()

// MARK: - TimeTrackingManager (With Tracking Modes, Day Boundaries Removed)

import Foundation
import SwiftData
import Combine
import UIKit
import UserNotifications

@MainActor
class TimeTrackingManager: ObservableObject {
    
    // MARK: - Properties
    
    private var modelContext: ModelContext
    
    // Published properties for UI updates
    @Published var currentTimeBlock: TimeBlock?
    @Published var activeSet: ActivitySet?
    @Published var currentTimeBlockRule: TimeBlockRule?
    
    // Background tracking
    private var backgroundEnteredTime: Date?
    private(set) var maxBackgroundDuration: TimeInterval = 900
    
    // Default tracking mode (when not within any time block rule)
    private(set) var defaultTrackingMode: TrackingMode = .on
    
    // Notification preferences
    var notificationsEnabled: Bool {
        get { UserDefaults.standard.bool(forKey: "notificationsEnabled") }
        set { UserDefaults.standard.set(newValue, forKey: "notificationsEnabled") }
    }
    
    // Timer for UI updates
    private var timer: Timer?
    
    // MARK: - Initialization
    
    init(modelContext: ModelContext) {
        self.modelContext = modelContext
        
        // Set default notification preference if not set
        if UserDefaults.standard.object(forKey: "notificationsEnabled") == nil {
            notificationsEnabled = true
        }
        
        loadSettings()
        resumeActiveTimeBlock()
        startTimer()
        setupBackgroundTracking()
        checkAndApplyCurrentTimeBlockRule()
    }
    
    // MARK: - Settings Management
    
    private func loadSettings() {
        loadBackgroundSettings()
        loadDefaultTrackingMode()
    }
    
    private func loadBackgroundSettings() {
        let savedDuration = UserDefaults.standard.double(forKey: "maxBackgroundDuration")
        if savedDuration > 0 {
            maxBackgroundDuration = savedDuration
        }
        
        if let savedTimestamp = UserDefaults.standard.object(forKey: "backgroundEnteredTime") as? Double {
            backgroundEnteredTime = Date(timeIntervalSince1970: savedTimestamp)
        }
    }
    
    private func loadDefaultTrackingMode() {
        if let savedMode = UserDefaults.standard.string(forKey: "defaultTrackingMode"),
           let mode = TrackingMode(rawValue: savedMode) {
            defaultTrackingMode = mode
        }
    }
    
    func saveMaxBackgroundDuration(_ duration: TimeInterval) {
        maxBackgroundDuration = duration
        UserDefaults.standard.set(duration, forKey: "maxBackgroundDuration")
    }
    
    func saveDefaultTrackingMode(_ mode: TrackingMode) {
        defaultTrackingMode = mode
        UserDefaults.standard.set(mode.rawValue, forKey: "defaultTrackingMode")
    }
    
    private func saveBackgroundState() {
        UserDefaults.standard.set(
            backgroundEnteredTime?.timeIntervalSince1970,
            forKey: "backgroundEnteredTime"
        )
    }
    
    // MARK: - Current Tracking Mode
    
    /// Get the current tracking mode based on active time block rule or default
    var currentTrackingMode: TrackingMode {
        // Find active time block rule
        if let activeRule = getActiveTimeBlockRule() {
            return activeRule.trackingMode
        }
        // Fall back to default
        return defaultTrackingMode
    }
    
    /// Whether gaps should be tracked right now
    var shouldTrackGaps: Bool {
        currentTrackingMode.tracksGaps
    }
    
    /// Whether auto-switching sets is enabled right now
    var shouldAutoSwitchSets: Bool {
        currentTrackingMode.autoSwitchesSets
    }
    
    /// Get the currently active time block rule (if any)
    func getActiveTimeBlockRule() -> TimeBlockRule? {
        let descriptor = FetchDescriptor<TimeBlockRule>(
            sortBy: [SortDescriptor(\.startTime, order: .forward)]
        )
        
        guard let rules = try? modelContext.fetch(descriptor) else { return nil }
        return rules.first(where: { $0.isActiveNow() })
    }
    
    // MARK: - Activity Management
    
    func startActivity(_ activity: Activity) {
        let now = Date()
        
        endCurrentActivityIfNeeded(at: now, state: "switched")
        
        let newBlock = TimeBlock(activity: activity, startTime: now)
        modelContext.insert(newBlock)
        currentTimeBlock = newBlock
        
        clearBackgroundState()
        saveContext()
    }
    
    func pauseCurrentActivity() {
        guard let current = currentTimeBlock, !current.isPaused else { return }
        
        let now = Date()
        
        current.endTime = now
        current.stopState = "paused"
        
        let pauseBlock = TimeBlock.createPause(startTime: now)
        modelContext.insert(pauseBlock)
        currentTimeBlock = pauseBlock
        
        clearBackgroundState()
        saveContext()
    }
    
    func resumeFromPause(_ activity: Activity) {
        guard let current = currentTimeBlock, current.isPauseBlock else { return }
        
        let now = Date()
        
        current.endTime = now
        current.stopState = "switched"
        current.switchCount += 1
        
        let resumedBlock = TimeBlock(activity: activity, startTime: now)
        modelContext.insert(resumedBlock)
        currentTimeBlock = resumedBlock
        
        saveContext()
    }
    
    func stopCurrentActivity() {
        guard let current = currentTimeBlock else { return }
        
        let now = Date()
        
        current.endTime = now
        current.stopState = "stopped"
        
        // Only create gap if current tracking mode allows it
        if shouldTrackGaps {
            let gap = TimeBlock.createGap(startTime: now)
            modelContext.insert(gap)
            currentTimeBlock = gap
        } else {
            // No gap tracking - just end the activity
            currentTimeBlock = nil
        }
        
        clearBackgroundState()
        saveContext()
    }
    
    private func endCurrentActivityIfNeeded(at time: Date, state: String) {
        guard let current = currentTimeBlock else { return }
        current.endTime = time
        current.stopState = state
    }
    
    private func clearBackgroundState() {
        backgroundEnteredTime = nil
        UserDefaults.standard.removeObject(forKey: "backgroundEnteredTime")
    }
    
    // MARK: - Timer Management
    
    private func startTimer() {
        timer = Timer.scheduledTimer(withTimeInterval: 1.0, repeats: true) { [weak self] _ in
            guard let self = self else { return }
            Task { @MainActor [weak self] in
                self?.objectWillChange.send()
                // Periodically check if time block rule has changed
                self?.checkAndApplyCurrentTimeBlockRule()
            }
        }
    }
    
    func stopTimer() {
        timer?.invalidate()
        timer = nil
    }
    
    // MARK: - Background Tracking
    
    private func setupBackgroundTracking() {
        NotificationCenter.default.addObserver(
            self,
            selector: #selector(appDidEnterBackground),
            name: UIApplication.didEnterBackgroundNotification,
            object: nil
        )
        
        NotificationCenter.default.addObserver(
            self,
            selector: #selector(appWillEnterForeground),
            name: UIApplication.willEnterForegroundNotification,
            object: nil
        )
    }
    
    @objc private func appDidEnterBackground() {
            // 1. Force reload settings to ensure we have the latest duration
            loadBackgroundSettings()
            
            guard let current = currentTimeBlock,
                  !current.isPaused,
                  !current.isGap else { return }
            
            backgroundEnteredTime = Date()
            saveBackgroundState()
            scheduleBackgroundReminder()
        }
    
    @objc private func appWillEnterForeground() {
        defer {
            backgroundEnteredTime = nil
            cancelBackgroundReminder()
        }
        
        guard let backgroundTime = backgroundEnteredTime,
              let current = currentTimeBlock,
              !current.isPaused else { return }
        
        let elapsedTime = Date().timeIntervalSince(backgroundTime)
        
        if elapsedTime > maxBackgroundDuration {
            handleBackgroundTimeout(
                current: current,
                backgroundTime: backgroundTime
            )
        }
        
        saveContext()
        checkAndApplyCurrentTimeBlockRule()
    }
    
    private func handleBackgroundTimeout(current: TimeBlock, backgroundTime: Date) {
        let cappedEndTime = backgroundTime.addingTimeInterval(maxBackgroundDuration)
        current.endTime = cappedEndTime
        current.stopState = "stopped"
        
        // Check tracking mode at the capped end time to decide about gap
        // For simplicity, use current tracking mode
        if shouldTrackGaps {
            let gap = TimeBlock.createGap(startTime: cappedEndTime)
            modelContext.insert(gap)
            currentTimeBlock = gap
        } else {
            currentTimeBlock = nil
        }
        
        notifyActivityStopped()
    }
    
    // MARK: - Session Recovery
    
    private func resumeActiveTimeBlock() {
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { $0.endTime == nil },
            sortBy: [SortDescriptor(\.startTime, order: .reverse)]
        )
        
        guard let activeBlock = try? modelContext.fetch(descriptor).first else { return }
        
        currentTimeBlock = activeBlock
        
        // Check if we need to handle a background timeout from previous session
        if let backgroundTime = backgroundEnteredTime {
            let elapsedTime = Date().timeIntervalSince(backgroundTime)
            
            if elapsedTime > maxBackgroundDuration {
                handleBackgroundTimeout(
                    current: activeBlock,
                    backgroundTime: backgroundTime
                )
                clearBackgroundState()
                saveContext()
            }
        }
    }
    
    // MARK: - Time Block Rule Application
        /// Check current time block rule and apply its settings
        func checkAndApplyCurrentTimeBlockRule() {
            let activeRule = getActiveTimeBlockRule()
            
            // Update current rule reference
            let previousRule = currentTimeBlockRule
            currentTimeBlockRule = activeRule
            
            // Only auto-switch if the mode allows it
            guard let rule = activeRule,
                  rule.trackingMode.autoSwitchesSets, // CHANGED: Access via trackingMode
                  let setID = rule.linkedSetID else { return }
            
            // Check if rule changed (to avoid repeated switches)
            if previousRule?.id != rule.id {
                let sets = try? modelContext.fetch(FetchDescriptor<ActivitySet>())
                let currentActiveSet = sets?.first { $0.isActive }
                
                if currentActiveSet?.id != setID,
                   let targetSet = sets?.first(where: { $0.id == setID }) {
                    switchToActivitySet(targetSet, ruleName: rule.name)
                }
            }
        }
    
    private func switchToActivitySet(_ set: ActivitySet, ruleName: String) {
        let sets = try? modelContext.fetch(FetchDescriptor<ActivitySet>())
        sets?.forEach { $0.isActive = false }
        set.isActive = true
        activeSet = set
        
        saveContext()
        notifySetSwitch(setName: set.name, ruleName: ruleName)
    }
    
    // MARK: - Notifications
    
    private func scheduleBackgroundReminder() {
        guard notificationsEnabled else { return }
        guard maxBackgroundDuration > 300 else { return }
        
        let reminderTime = maxBackgroundDuration - 300
        
        let content = UNMutableNotificationContent()
        content.title = "Cadence Activity Tracking"
        content.body = "Your activity will stop in 5 minutes. Open the app to continue tracking."
        content.sound = .default
        
        let trigger = UNTimeIntervalNotificationTrigger(
            timeInterval: reminderTime,
            repeats: false
        )
        let request = UNNotificationRequest(
            identifier: "backgroundReminder",
            content: content,
            trigger: trigger
        )
        
        UNUserNotificationCenter.current().add(request) { error in
            if let error = error {
                print("Error scheduling reminder: \(error)")
            }
        }
    }
    
    private func cancelBackgroundReminder() {
        UNUserNotificationCenter.current().removePendingNotificationRequests(
            withIdentifiers: ["backgroundReminder"]
        )
    }
    
    private func notifyActivityStopped() {
        guard notificationsEnabled else { return }
        
        let content = UNMutableNotificationContent()
        content.title = "Activity Stopped"
        content.body = "Your activity was stopped after \(maxBackgroundDuration.formattedDuration) in background."
        content.sound = .default
        
        let trigger = UNTimeIntervalNotificationTrigger(timeInterval: 1, repeats: false)
        let request = UNNotificationRequest(
            identifier: "activityStopped",
            content: content,
            trigger: trigger
        )
        
        UNUserNotificationCenter.current().add(request) { error in
            if let error = error {
                print("Error showing notification: \(error)")
            }
        }
    }
    
    private func notifySetSwitch(setName: String, ruleName: String) {
        guard notificationsEnabled else { return }
        
        let content = UNMutableNotificationContent()
        content.title = "Activity Set Changed"
        content.body = "\(ruleName): Switched to \(setName)"
        content.sound = .default
        
        let trigger = UNTimeIntervalNotificationTrigger(timeInterval: 1, repeats: false)
        let request = UNNotificationRequest(
            identifier: "setSwitch",
            content: content,
            trigger: trigger
        )
        
        UNUserNotificationCenter.current().add(request) { error in
            if let error = error {
                print("Error showing notification: \(error)")
            }
        }
    }
    
    // MARK: - Data Queries
    
    func getTodaysTimeBlocks() -> [TimeBlock] {
        let calendar = Calendar.current
        let startOfDay = calendar.startOfDay(for: Date())
        let endOfDay = calendar.date(byAdding: .day, value: 1, to: startOfDay)!
        
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { block in
                block.startTime >= startOfDay && block.startTime < endOfDay
            },
            sortBy: [SortDescriptor(\.startTime, order: .forward)]
        )
        
        return (try? modelContext.fetch(descriptor)) ?? []
    }
    
    func getTodaysStats() -> DayStats {
        let blocks = getTodaysTimeBlocks()
        
        var activityDurations: [String: TimeInterval] = [:]
        var totalDuration: TimeInterval = 0
        var totalSwitches: Int = 0
        
        for block in blocks {
            let duration = block.duration
            activityDurations[block.activityName, default: 0] += duration
            totalDuration += duration
            totalSwitches += block.switchCount
        }
        
        return DayStats(
            totalDuration: totalDuration,
            activityDurations: activityDurations,
            totalSwitches: totalSwitches,
            blockCount: blocks.count
        )
    }
    
    // MARK: - Helper Methods
    
    private func saveContext() {
        try? modelContext.save()
    }
}

// MARK: - Stats Model

struct DayStats {
    let totalDuration: TimeInterval
    let activityDurations: [String: TimeInterval]
    let totalSwitches: Int
    let blockCount: Int
    
    var sortedActivities: [(name: String, duration: TimeInterval)] {
        activityDurations
            .sorted { $0.value > $1.value }
            .map { (name: $0.key, duration: $0.value) }
    }
}

//

import Foundation
import SwiftData

// MARK: - Archive Models (DTOs)
// These structs ensure data is portable and independent of SwiftData context
struct BackupData: Codable {
    let metadata: BackupMetadata
    let activities: [ActivityDTO]
    let activitySets: [ActivitySetDTO]
    let timeBlocks: [TimeBlockDTO]
}

struct BackupMetadata: Codable {
    var version: String = "1.0"
    let exportDate: Date
    let recordCount: Int
    let appVersion: String
}

struct ActivityDTO: Codable {
    let id: UUID
    let name: String
    let emoji: String?
    let color: String
    let setID: UUID
    let sortOrder: Int
}

struct ActivitySetDTO: Codable {
    let id: UUID
    let name: String
    let isActive: Bool
}

struct TimeBlockDTO: Codable {
    let id: UUID
    let activityID: UUID?
    let activityName: String
    let activityColor: String
    let startTime: Date
    let endTime: Date?
    let remarks: String
    let blockType: String
    let duration: TimeInterval
    let switchCount: Int
    let totalPauseDuration: TimeInterval
}

// MARK: - Manager Class
class DataArchiveManager {
    let modelContext: ModelContext
    
    init(modelContext: ModelContext) {
        self.modelContext = modelContext
    }
    
    // MARK: - Archive (Export JSON)
    func createArchive(olderThan date: Date? = nil) throws -> URL {
        // 1. Fetch Data
        // If date is provided, filter. If nil, export all.
        let blockDescriptor = FetchDescriptor<TimeBlock>(
            predicate: date != nil ? #Predicate<TimeBlock> { $0.startTime < date! } : nil
        )
        let blocks = try modelContext.fetch(blockDescriptor)
        
        let activities = try modelContext.fetch(FetchDescriptor<Activity>())
        let sets = try modelContext.fetch(FetchDescriptor<ActivitySet>())
        
        // 2. Map to DTOs
        let blockDTOs = blocks.map { block in
            TimeBlockDTO(
                id: block.id,
                activityID: block.activityID,
                activityName: block.activityName,
                activityColor: block.activityColor,
                startTime: block.startTime,
                endTime: block.endTime,
                remarks: block.remarks,
                blockType: block.blockType,
                duration: block.duration,
                switchCount: block.switchCount,
                totalPauseDuration: block.totalPauseDuration
            )
        }
        
        let activityDTOs = activities.map { act in
            ActivityDTO(id: act.id, name: act.name, emoji: act.emoji, color: act.color, setID: act.setID, sortOrder: act.sortOrder)
        }
        
        let setDTOs = sets.map { set in
            ActivitySetDTO(id: set.id, name: set.name, isActive: set.isActive)
        }
        
        // 3. Create Backup Object
        let metadata = BackupMetadata(
            exportDate: Date(),
            recordCount: blockDTOs.count,
            appVersion: Bundle.main.infoDictionary?["CFBundleShortVersionString"] as? String ?? "1.0"
        )
        
        let backup = BackupData(
            metadata: metadata,
            activities: activityDTOs,
            activitySets: setDTOs,
            timeBlocks: blockDTOs
        )
        
        // 4. Encode and Write
        let encoder = JSONEncoder()
        encoder.dateEncodingStrategy = .iso8601
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        
        let data = try encoder.encode(backup)
        
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd_HHmm"
        let timestamp = formatter.string(from: Date())
        let filename = "Cadence_Archive_\(timestamp).json"
        
        let url = FileManager.default.temporaryDirectory.appendingPathComponent(filename)
        try data.write(to: url)
        
        return url
    }
    
    // MARK: - Restore (Import JSON)
    func restoreFromArchive(url: URL) throws -> Int {
        // 1. Decode
        // Access security scoped resource if necessary (for picking files outside sandbox)
        let accessing = url.startAccessingSecurityScopedResource()
        defer { if accessing { url.stopAccessingSecurityScopedResource() } }
        
        let data = try Data(contentsOf: url)
        let decoder = JSONDecoder()
        decoder.dateDecodingStrategy = .iso8601
        
        let backup = try decoder.decode(BackupData.self, from: data)
        var restoredCount = 0
        
        // 2. Insert Data
        // Strategy: We strictly add data. We do not delete existing data.
        // We check for ID collisions to prevent duplicates.
        
        // A. Restore Sets
        let existingSetIDs = try Set(modelContext.fetch(FetchDescriptor<ActivitySet>()).map { $0.id })
        for setDTO in backup.activitySets {
            if !existingSetIDs.contains(setDTO.id) {
                let newSet = ActivitySet(name: setDTO.name, isActive: setDTO.isActive)
                newSet.id = setDTO.id // Preserve ID
                modelContext.insert(newSet)
            }
        }
        
        // B. Restore Activities
        let existingActivityIDs = try Set(modelContext.fetch(FetchDescriptor<Activity>()).map { $0.id })
        for actDTO in backup.activities {
            if !existingActivityIDs.contains(actDTO.id) {
                let newActivity = Activity(
                    name: actDTO.name,
                    color: actDTO.color,
                    setID: actDTO.setID,
                    sortOrder: actDTO.sortOrder,
                    emoji: actDTO.emoji
                )
                newActivity.id = actDTO.id // Preserve ID
                modelContext.insert(newActivity)
            }
        }
        
        // C. Restore TimeBlocks
        // This can be slow for large datasets, so we do it in a batch if possible
        let existingBlockIDs = try Set(modelContext.fetch(FetchDescriptor<TimeBlock>()).map { $0.id })
        
        for blockDTO in backup.timeBlocks {
            if !existingBlockIDs.contains(blockDTO.id) {
                let newBlock = TimeBlock(
                    activityID: blockDTO.activityID,
                    activityName: blockDTO.activityName,
                    activityColor: blockDTO.activityColor,
                    activitySetID: nil, // We don't strictly need to link this for archives
                    startTime: blockDTO.startTime,
                    blockType: blockDTO.blockType
                )
                newBlock.id = blockDTO.id
                newBlock.endTime = blockDTO.endTime
                newBlock.remarks = blockDTO.remarks
                newBlock.switchCount = blockDTO.switchCount
                newBlock.totalPauseDuration = blockDTO.totalPauseDuration
                newBlock.stopState = blockDTO.endTime != nil ? "stopped" : "ongoing"
                
                modelContext.insert(newBlock)
                restoredCount += 1
            }
        }
        
        try modelContext.save()
        return restoredCount
    }
    
    // MARK: - Prune
    func deleteRecords(olderThan date: Date) throws -> Int {
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate<TimeBlock> { $0.startTime < date }
        )
        let count = try modelContext.fetchCount(descriptor)
        
        try modelContext.delete(model: TimeBlock.self, where: #Predicate<TimeBlock> { $0.startTime < date })
        try modelContext.save()
        
        return count
    }
}

//

import Foundation
import SwiftData

// MARK: - Top Level Structs (Moved out to fix inference errors)
struct ActivityChange: Identifiable, Hashable {
    var id: String { name } // Conforms to Identifiable
    let name: String
    let change: TimeInterval
    let percentageChange: Double
    let goal: ActivityGoal
}

class ReportGenerator {
    
    // MARK: - Report Types
    enum ReportPeriod: Hashable, Equatable {
        case thisWeek, lastWeek, thisMonth, lastMonth
        case custom(start: Date, end: Date)
        
        var dateRange: (start: Date, end: Date) {
            let calendar = Calendar.current
            let now = Date()
            switch self {
            case .thisWeek:
                let start = calendar.date(from: calendar.dateComponents([.yearForWeekOfYear, .weekOfYear], from: now))!
                return (start, calendar.date(byAdding: .day, value: 7, to: start)!)
            case .lastWeek:
                let start = calendar.date(from: calendar.dateComponents([.yearForWeekOfYear, .weekOfYear], from: now))!
                let lastStart = calendar.date(byAdding: .weekOfYear, value: -1, to: start)!
                return (lastStart, start)
            case .thisMonth:
                let start = calendar.date(from: calendar.dateComponents([.year, .month], from: now))!
                return (start, calendar.date(byAdding: .month, value: 1, to: start)!)
            case .lastMonth:
                let start = calendar.date(from: calendar.dateComponents([.year, .month], from: now))!
                let lastStart = calendar.date(byAdding: .month, value: -1, to: start)!
                return (lastStart, start)
            case .custom(let s, let e): return (s, e)
            }
        }
        
        var displayName: String {
            switch self {
            case .thisWeek: return "This Week"
            case .lastWeek: return "Last Week"
            case .thisMonth: return "This Month"
            case .lastMonth: return "Last Month"
            case .custom: return "Custom"
            }
        }
    }
    
    // MARK: - Report Structure
    struct Report {
        let period: ReportPeriod
        let totalDuration: TimeInterval
        let totalBlocks: Int
        let totalSwitches: Int
        let activityBreakdown: [(name: String, duration: TimeInterval, percentage: Double, color: String)]
        let dailyTotals: [(date: Date, duration: TimeInterval)]
        let comparisonToPrevious: ComparisonData?
        let topActivities: [(name: String, duration: TimeInterval)]
        let mostProductiveDay: (date: Date, duration: TimeInterval)?
        
        struct ComparisonData {
            let previousTotalDuration: TimeInterval
            let durationChange: TimeInterval
            let percentageChange: Double
            let activityChanges: [ActivityChange] // Uses the top-level struct
        }
    }
    
    // MARK: - Generate Report
    func generateReport(for period: ReportPeriod, blocks: [TimeBlock], activities: [Activity]) -> Report {
        let (start, end) = period.dateRange
        let periodBlocks = blocks.filter { $0.startTime >= start && $0.startTime < end }
        
        let totalDuration = periodBlocks.reduce(0) { $0 + $1.duration }
        let totalSwitches = periodBlocks.reduce(0) { $0 + $1.switchCount }
        
        // Breakdown
        var durMap: [String: (TimeInterval, String)] = [:]
        for b in periodBlocks {
            let curr = durMap[b.activityName] ?? (0, b.activityColor)
            durMap[b.activityName] = (curr.0 + b.duration, b.activityColor)
        }
        
        let breakdown = durMap.map { k, v in
            (name: k, duration: v.0, percentage: totalDuration > 0 ? (v.0/totalDuration)*100 : 0, color: v.1)
        }.sorted { $0.duration > $1.duration }
        
        // Daily
        let cal = Calendar.current
        var dailyMap: [Date: TimeInterval] = [:]
        for b in periodBlocks {
            let d = cal.startOfDay(for: b.startTime)
            dailyMap[d, default: 0] += b.duration
        }
        let dailyTotals = dailyMap.sorted { $0.key < $1.key }.map { ($0.key, $0.value) }
        
        let top5 = Array(breakdown.prefix(5)).map { ($0.name, $0.duration) }
        let bestDay = dailyTotals.max { $0.1 < $1.1 }
        
        // Goal Lookup
        let goalMap = Dictionary(activities.map { ($0.name, $0.goal) }, uniquingKeysWith: { (first, _) in first })
        
        let comparison = calculateComparison(currentBlocks: periodBlocks, period: period, allBlocks: blocks, goalMap: goalMap)
        
        return Report(
            period: period,
            totalDuration: totalDuration,
            totalBlocks: periodBlocks.count,
            totalSwitches: totalSwitches,
            activityBreakdown: breakdown,
            dailyTotals: dailyTotals,
            comparisonToPrevious: comparison,
            topActivities: top5,
            mostProductiveDay: bestDay
        )
    }
    
    private func calculateComparison(currentBlocks: [TimeBlock], period: ReportPeriod, allBlocks: [TimeBlock], goalMap: [String: ActivityGoal]) -> Report.ComparisonData? {
        let (cStart, cEnd) = period.dateRange
        let pEnd = cStart
        let pStart: Date
        
        // Calculate previous start
        switch period {
        case .thisWeek, .lastWeek:
            pStart = cStart.addingTimeInterval(-(cEnd.timeIntervalSince(cStart)))
        case .thisMonth, .lastMonth:
            pStart = Calendar.current.date(byAdding: .month, value: -1, to: cStart)!
        case .custom:
            pStart = cStart.addingTimeInterval(-(cEnd.timeIntervalSince(cStart)))
        }
        
        let prevBlocks = allBlocks.filter { $0.startTime >= pStart && $0.startTime < pEnd }
        if prevBlocks.isEmpty { return nil }
        
        let prevDur = prevBlocks.reduce(0) { $0 + $1.duration }
        let currDur = currentBlocks.reduce(0) { $0 + $1.duration }
        let diff = currDur - prevDur
        
        // Activity Changes
        var cMap: [String: TimeInterval] = [:]
        var pMap: [String: TimeInterval] = [:]
        for b in currentBlocks { cMap[b.activityName, default: 0] += b.duration }
        for b in prevBlocks { pMap[b.activityName, default: 0] += b.duration }
        
        let allNames = Set(cMap.keys).union(pMap.keys)
        let changes: [ActivityChange] = allNames.map { name in
            let c = cMap[name] ?? 0
            let p = pMap[name] ?? 0
            let chg = c - p
            let pct = p > 0 ? (chg/p)*100 : (c > 0 ? 100 : 0)
            return ActivityChange(name: name, change: chg, percentageChange: pct, goal: goalMap[name] ?? .neutral)
        }.sorted { abs($0.change) > abs($1.change) }
        
        return Report.ComparisonData(
            previousTotalDuration: prevDur,
            durationChange: diff,
            percentageChange: prevDur > 0 ? (diff/prevDur)*100 : 0,
            activityChanges: changes
        )
    }
    
    func formatReportAsText(_ report: Report) -> String {
        return "Report for \(report.period.displayName)\nTotal: \(report.totalDuration.formattedDuration)"
    }
}

import Foundation
import SwiftData

class CSVExporter {
    
    // MARK: - Export Configuration
    struct ExportOptions {
        var includeActivity: Bool = true
        var includeStartTime: Bool = true
        var includeEndTime: Bool = true
        var includeDuration: Bool = true
        var includePauseDuration: Bool = true
        var includeSwitchCount: Bool = true
        var includeRemarks: Bool = true
    }
    
    // MARK: - Export Time Blocks
    func exportTimeBlocks(_ blocks: [TimeBlock], options: ExportOptions = ExportOptions()) -> String {
        var csv = generateHeader(options: options) + "\n"
        
        for block in blocks {
            var row: [String] = []
            
            if options.includeActivity {
                row.append(escapeCSV(block.activityName))
            }
            
            if options.includeStartTime {
                row.append(formatDateTime(block.startTime))
            }
            
            if options.includeEndTime {
                let endTimeStr = block.endTime.map { formatDateTime($0) } ?? "Ongoing"
                row.append(endTimeStr)
            }
            
            if options.includeDuration {
                row.append(formatDuration(block.duration))
            }
            
            if options.includePauseDuration {
                row.append(formatDuration(block.totalPauseDuration))
            }
            
            if options.includeSwitchCount {
                row.append("\(block.switchCount)")
            }
            
            if options.includeRemarks {
                row.append(escapeCSV(block.remarks))
            }
            
            csv += row.joined(separator: ",") + "\n"
        }
        
        return csv
    }
    
    // MARK: - Generate Header
    private func generateHeader(options: ExportOptions) -> String {
        var headers: [String] = []
        
        if options.includeActivity { headers.append("Activity") }
        if options.includeStartTime { headers.append("Start Time") }
        if options.includeEndTime { headers.append("End Time") }
        if options.includeDuration { headers.append("Duration") }
        if options.includePauseDuration { headers.append("Pause Duration") }
        if options.includeSwitchCount { headers.append("Switches") }
        if options.includeRemarks { headers.append("Remarks") }
        
        return headers.joined(separator: ",")
    }
    
    // MARK: - Formatting Helpers
    private func formatDateTime(_ date: Date) -> String {
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd HH:mm:ss"
        return formatter.string(from: date)
    }
    
    private func formatDuration(_ duration: TimeInterval) -> String {
        let hours = Int(duration) / 3600
        let minutes = Int(duration) / 60 % 60
        let seconds = Int(duration) % 60
        
        if hours > 0 {
            return String(format: "%dh %dm %ds", hours, minutes, seconds)
        } else if minutes > 0 {
            return String(format: "%dm %ds", minutes, seconds)
        } else {
            return String(format: "%ds", seconds)
        }
    }
    
    private func escapeCSV(_ string: String) -> String {
        // If string contains comma, quote, or newline, wrap in quotes and escape quotes
        if string.contains(",") || string.contains("\"") || string.contains("\n") {
            let escaped = string.replacingOccurrences(of: "\"", with: "\"\"")
            return "\"\(escaped)\""
        }
        return string
    }
    
    // MARK: - Save to File
    func saveToFile(csv: String, filename: String) throws -> URL {
        let documentsPath = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
        let fileURL = documentsPath.appendingPathComponent(filename)
        
        try csv.write(to: fileURL, atomically: true, encoding: .utf8)
        
        return fileURL
    }
    
    // MARK: - Generate Filename
    func generateFilename(prefix: String = "cadence_export") -> String {
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd_HHmm"
        let timestamp = formatter.string(from: Date())
        return "\(prefix)_\(timestamp).csv"
    }
}

//
//  TimeBlockEditor.swift
//  Cadence
//
//  Created by Zhi Zheng Yeo on 30/1/26.
//  Updated: Fixed merge/split logic - now allows splitting any activity


import Foundation
import SwiftData

@MainActor
class TimeBlockEditor {
    private var modelContext: ModelContext
    
    init(modelContext: ModelContext) {
        self.modelContext = modelContext
    }
    
    // MARK: - Validation
    
    /// Check if two blocks are truly adjacent (end time of first matches start time of second)
    func areAdjacent(_ block1: TimeBlock, _ block2: TimeBlock) -> Bool {
        guard let end1 = block1.endTime else { return false }
        let timeDiff = abs(end1.timeIntervalSince(block2.startTime))
        return timeDiff < 1.0 // Within 1 second tolerance
    }
    
    /// Check if a set of blocks can be merged
    func canMerge(_ blocks: [TimeBlock]) -> Bool {
        guard blocks.count >= 2 else { return false }
        
        let sorted = blocks.sorted { $0.startTime < $1.startTime }
        
        // Check all blocks are adjacent
        for i in 0..<(sorted.count - 1) {
            if !areAdjacent(sorted[i], sorted[i + 1]) {
                return false
            }
        }
        
        return true
    }
    
    /// Check if a block can be split
    func canSplit(_ block: TimeBlock) -> Bool {
        // Can split any completed block
        return block.endTime != nil
    }
    
    // MARK: - Merge
    
    func mergeBlocks(_ blocks: [TimeBlock], intoActivity targetActivity: Activity?) throws {
        guard canMerge(blocks) else {
            throw TimeBlockEditorError.cannotMerge
        }
        
        let sorted = blocks.sorted { $0.startTime < $1.startTime }
        
        guard let firstBlock = sorted.first,
              let lastBlock = sorted.last,
              let endTime = lastBlock.endTime else {
            throw TimeBlockEditorError.invalidBlocks
        }
        
        // Create the merged block
        let mergedBlock: TimeBlock
        if let activity = targetActivity {
            mergedBlock = TimeBlock(activity: activity, startTime: firstBlock.startTime)
        } else {
            mergedBlock = TimeBlock.createGap(startTime: firstBlock.startTime)
        }
        
        mergedBlock.endTime = endTime
        mergedBlock.stopState = lastBlock.stopState
        
        // Concatenate remarks with separator
        let allRemarks = sorted.compactMap { $0.remarks.isEmpty ? nil : $0.remarks }
        mergedBlock.remarks = allRemarks.joined(separator: "\n---\n")
        
        // Sum up pause durations and switch counts
        mergedBlock.totalPauseDuration = sorted.reduce(0) { $0 + $1.totalPauseDuration }
        mergedBlock.switchCount = sorted.reduce(0) { $0 + $1.switchCount }
        
        // Delete original blocks
        for block in sorted {
            modelContext.delete(block)
        }
        
        // Insert merged block
        modelContext.insert(mergedBlock)
        try modelContext.save()
    }
    
    // MARK: - Split
    
    /// Split a block at the given time
    /// - For activities: both parts become the same activity
    /// - For gaps/pauses: both parts remain as gaps/pauses
    func splitBlock(_ block: TimeBlock, at splitTime: Date) throws {
        guard canSplit(block) else {
            throw TimeBlockEditorError.cannotSplit
        }
        
        guard let endTime = block.endTime else {
            throw TimeBlockEditorError.blockStillActive
        }
        
        guard splitTime > block.startTime && splitTime < endTime else {
            throw TimeBlockEditorError.invalidSplitTime
        }
        
        // Create first block (same type as original)
        let firstBlock: TimeBlock
        if let activityID = block.activityID {
            let descriptor = FetchDescriptor<Activity>(
                predicate: #Predicate { $0.id == activityID }
            )
            if let activity = try? modelContext.fetch(descriptor).first {
                firstBlock = TimeBlock(activity: activity, startTime: block.startTime)
            } else {
                throw TimeBlockEditorError.activityNotFound
            }
        } else if block.isGap {
            firstBlock = TimeBlock.createGap(startTime: block.startTime)
        } else {
            firstBlock = TimeBlock.createPause(startTime: block.startTime)
        }
        
        firstBlock.endTime = splitTime
        firstBlock.stopState = "switched"
        firstBlock.remarks = block.remarks
        
        // Create second block (same type as original - NOT a gap anymore)
        let secondBlock: TimeBlock
        if let activityID = block.activityID {
            let descriptor = FetchDescriptor<Activity>(
                predicate: #Predicate { $0.id == activityID }
            )
            if let activity = try? modelContext.fetch(descriptor).first {
                secondBlock = TimeBlock(activity: activity, startTime: splitTime)
            } else {
                throw TimeBlockEditorError.activityNotFound
            }
        } else if block.isGap {
            secondBlock = TimeBlock.createGap(startTime: splitTime)
        } else {
            secondBlock = TimeBlock.createPause(startTime: splitTime)
        }
        
        secondBlock.endTime = endTime
        secondBlock.stopState = block.stopState
        // Don't copy remarks to second block - they stay with first
        
        // Delete original and insert new blocks
        modelContext.delete(block)
        modelContext.insert(firstBlock)
        modelContext.insert(secondBlock)
        
        try modelContext.save()
    }
    
    // MARK: - Convert
    
    func convertToActivity(_ block: TimeBlock, activity: Activity) throws {
        guard block.isGap || block.isPauseBlock else {
            throw TimeBlockEditorError.notAGap
        }
        
        block.activityID = activity.id
        block.activityName = activity.name
        block.activityColor = activity.color
        block.activitySetID = activity.setID
        block.blockType = "activity"
        
        try modelContext.save()
    }
    
    // MARK: - Helpers
    
    func getAdjacentBlocks(for block: TimeBlock) -> (previous: TimeBlock?, next: TimeBlock?) {
        let calendar = Calendar.current
        let dayStart = calendar.startOfDay(for: block.startTime)
        let dayEnd = calendar.date(byAdding: .day, value: 1, to: dayStart)!
        
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { b in
                b.startTime >= dayStart && b.startTime < dayEnd
            },
            sortBy: [SortDescriptor(\.startTime, order: .forward)]
        )
        
        guard let allBlocks = try? modelContext.fetch(descriptor),
              let currentIndex = allBlocks.firstIndex(where: { $0.id == block.id }) else {
            return (nil, nil)
        }
        
        // Check for truly adjacent previous block
        var previous: TimeBlock? = nil
        if currentIndex > 0 {
            let prev = allBlocks[currentIndex - 1]
            if areAdjacent(prev, block) {
                previous = prev
            }
        }
        
        // Check for truly adjacent next block
        var next: TimeBlock? = nil
        if currentIndex < allBlocks.count - 1 {
            let nextBlock = allBlocks[currentIndex + 1]
            if areAdjacent(block, nextBlock) {
                next = nextBlock
            }
        }
        
        return (previous, next)
    }
}

enum TimeBlockEditorError: LocalizedError {
    case cannotMerge, cannotSplit, invalidBlocks, blockStillActive
    case invalidSplitTime, blockTooLong, activityNotFound, notAGap, alreadyGap
    
    var errorDescription: String? {
        switch self {
        case .cannotMerge: return "Cannot merge these blocks. Blocks must be adjacent and at least one must be a gap, pause, or short activity (<5 min)."
        case .cannotSplit: return "Cannot split this block. Block must be completed and at least 2 minutes long."
        case .invalidBlocks: return "Invalid blocks selected."
        case .blockStillActive: return "Cannot split an active block."
        case .invalidSplitTime: return "Invalid split time."
        case .blockTooLong: return "Block is too long to modify."
        case .activityNotFound: return "Activity not found."
        case .notAGap: return "Only gaps and pauses can be converted to activities."
        case .alreadyGap: return "This block is already a gap."
        }
    }
}

//
//  ConflictDetector.swift
//  Cadence
//
//  Validates time block edits and detects conflicts
//

import Foundation
import SwiftData

// MARK: - Conflict Types

enum TimeConflict: Identifiable, Equatable {
    case overlap(with: TimeBlock, overlapMinutes: Int)
    case completeOverwrite(victims: [TimeBlock])
    case exceedsDayBoundary
    case endBeforeStart
    case zeroOrNegativeDuration
    
    var id: String {
        switch self {
        case .overlap(let block, _): return "overlap-\(block.id)"
        case .completeOverwrite(let victims): return "overwrite-\(victims.map { $0.id.uuidString }.joined())"
        case .exceedsDayBoundary: return "boundary"
        case .endBeforeStart: return "invalid-range"
        case .zeroOrNegativeDuration: return "zero-duration"
        }
    }
    
    var severity: ConflictSeverity {
        switch self {
        case .endBeforeStart, .zeroOrNegativeDuration: return .blocking
        case .completeOverwrite: return .destructive
        case .overlap: return .warning
        case .exceedsDayBoundary: return .warning
        }
    }
    
    var description: String {
        switch self {
        case .overlap(let block, let minutes):
            return "Overlaps with \"\(block.activityName)\" by \(minutes) minutes"
        case .completeOverwrite(let victims):
            let names = victims.map { $0.activityName }.joined(separator: ", ")
            return "This will delete: \(names)"
        case .exceedsDayBoundary:
            return "Time extends beyond the current day"
        case .endBeforeStart:
            return "End time must be after start time"
        case .zeroOrNegativeDuration:
            return "Block must have a positive duration"
        }
    }
}

enum ConflictSeverity {
    case warning     // Can proceed with confirmation
    case destructive // Requires explicit "I understand" confirmation
    case blocking    // Cannot proceed at all
}

// MARK: - Validation Result

struct ValidationResult {
    let isValid: Bool
    let conflicts: [TimeConflict]
    let canProceedWithConfirmation: Bool
    
    var blockingConflicts: [TimeConflict] {
        conflicts.filter { $0.severity == .blocking }
    }
    
    var destructiveConflicts: [TimeConflict] {
        conflicts.filter { $0.severity == .destructive }
    }
    
    var warnings: [TimeConflict] {
        conflicts.filter { $0.severity == .warning }
    }
    
    static var valid: ValidationResult {
        ValidationResult(isValid: true, conflicts: [], canProceedWithConfirmation: true)
    }
}

// MARK: - Conflict Detector

@MainActor
class ConflictDetector {
    private let modelContext: ModelContext
    
    init(modelContext: ModelContext) {
        self.modelContext = modelContext
    }
    
    // MARK: - Main Validation Entry Point
    
    /// Validate a proposed edit to a time block
    func validateEdit(
        block: TimeBlock,
        newStart: Date,
        newEnd: Date
    ) -> ValidationResult {
        var conflicts: [TimeConflict] = []
        
        // 1. Basic validity checks
        if newEnd <= newStart {
            conflicts.append(.endBeforeStart)
        }
        
        let duration = newEnd.timeIntervalSince(newStart)
        if duration <= 0 {
            conflicts.append(.zeroOrNegativeDuration)
        }
        
        // 2. Day boundary check
        let calendar = Calendar.current
        if !calendar.isDate(newStart, inSameDayAs: newEnd) {
            conflicts.append(.exceedsDayBoundary)
        }
        
        // If we have blocking conflicts, return early
        if conflicts.contains(where: { $0.severity == .blocking }) {
            return ValidationResult(
                isValid: false,
                conflicts: conflicts,
                canProceedWithConfirmation: false
            )
        }
        
        // 3. Find overlapping blocks
        let dayStart = calendar.startOfDay(for: newStart)
        let dayEnd = calendar.date(byAdding: .day, value: 1, to: dayStart)!
        
        let overlaps = findOverlappingBlocks(
            excluding: block.id,
            in: dayStart...dayEnd,
            proposedStart: newStart,
            proposedEnd: newEnd
        )
        
        // 4. Categorize overlaps
        for other in overlaps {
            let otherEnd = other.endTime ?? Date()
            
            // Check if this edit would completely overwrite the other block
            if newStart <= other.startTime && newEnd >= otherEnd {
                // Find ALL blocks that would be overwritten
                let victims = overlaps.filter { victim in
                    let victimEnd = victim.endTime ?? Date()
                    return newStart <= victim.startTime && newEnd >= victimEnd
                }
                if !victims.isEmpty {
                    conflicts.append(.completeOverwrite(victims: victims))
                }
                break // Don't double-count
            } else {
                // Partial overlap
                let overlapStart = max(newStart, other.startTime)
                let overlapEnd = min(newEnd, otherEnd)
                let overlapMinutes = Int(overlapEnd.timeIntervalSince(overlapStart) / 60)
                
                if overlapMinutes > 0 {
                    conflicts.append(.overlap(with: other, overlapMinutes: overlapMinutes))
                }
            }
        }
        
        // Remove duplicate overwrites
        let uniqueConflicts = deduplicateConflicts(conflicts)
        
        let hasBlocking = uniqueConflicts.contains { $0.severity == .blocking }
        let canProceed = !hasBlocking
        
        return ValidationResult(
            isValid: uniqueConflicts.isEmpty,
            conflicts: uniqueConflicts,
            canProceedWithConfirmation: canProceed
        )
    }
    
    // MARK: - Gap Detection
    
    /// Find all gaps in a given date range
    func findGaps(for date: Date) -> [TimeBlock] {
        let calendar = Calendar.current
        let dayStart = calendar.startOfDay(for: date)
        let dayEnd = calendar.date(byAdding: .day, value: 1, to: dayStart)!
        
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { block in
                block.startTime >= dayStart &&
                block.startTime < dayEnd &&
                block.blockType == "gap"
            },
            sortBy: [SortDescriptor(\.startTime)]
        )
        
        return (try? modelContext.fetch(descriptor)) ?? []
    }
    
    /// Count unresolved gaps for badge display
    func unresolvedGapCount(for date: Date? = nil) -> Int {
        if let date = date {
            return findGaps(for: date).count
        }
        
        // All unresolved gaps
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { $0.blockType == "gap" }
        )
        return (try? modelContext.fetchCount(descriptor)) ?? 0
    }
    
    // MARK: - Helpers
    
    private func findOverlappingBlocks(
        excluding blockID: UUID,
        in dateRange: ClosedRange<Date>,
        proposedStart: Date,
        proposedEnd: Date
    ) -> [TimeBlock] {
        let rangeStart = dateRange.lowerBound
        let rangeEnd = dateRange.upperBound
        
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { block in
                block.id != blockID &&
                block.startTime >= rangeStart &&
                block.startTime < rangeEnd
            },
            sortBy: [SortDescriptor(\.startTime)]
        )
        
        guard let blocks = try? modelContext.fetch(descriptor) else { return [] }
        
        // Filter to only those that actually overlap
        return blocks.filter { block in
            let blockEnd = block.endTime ?? Date()
            // Overlap exists if: proposedStart < blockEnd AND proposedEnd > blockStart
            return proposedStart < blockEnd && proposedEnd > block.startTime
        }
    }
    
    private func deduplicateConflicts(_ conflicts: [TimeConflict]) -> [TimeConflict] {
        var seen = Set<String>()
        return conflicts.filter { conflict in
            if seen.contains(conflict.id) {
                return false
            }
            seen.insert(conflict.id)
            return true
        }
    }
}

// MARK: - Suggested Resolutions

extension ConflictDetector {
    
    struct Resolution: Identifiable {
        let id = UUID()
        let title: String
        let description: String
        let action: () -> Void
    }
    
    /// Generate suggested resolutions for conflicts
    func suggestResolutions(
        for conflicts: [TimeConflict],
        block: TimeBlock,
        proposedStart: Date,
        proposedEnd: Date
    ) -> [Resolution] {
        var resolutions: [Resolution] = []
        
        for conflict in conflicts {
            switch conflict {
            case .overlap(let other, _):
                // Suggest: Trim this block to fit before the other
                resolutions.append(Resolution(
                    title: "End at \(other.startTime.formatted(date: .omitted, time: .shortened))",
                    description: "Adjust end time to avoid overlap",
                    action: {
                        // Caller handles the actual edit
                    }
                ))
                
                // Suggest: Trim this block to fit after the other
                if let otherEnd = other.endTime {
                    resolutions.append(Resolution(
                        title: "Start at \(otherEnd.formatted(date: .omitted, time: .shortened))",
                        description: "Adjust start time to avoid overlap",
                        action: {}
                    ))
                }
                
            case .completeOverwrite(let victims):
                resolutions.append(Resolution(
                    title: "Delete \(victims.count) block(s) and proceed",
                    description: "This cannot be undone",
                    action: {}
                ))
                
            default:
                break
            }
        }
        
        return resolutions
    }
}

// MARK: - Time Block Rule Model
@Model
final class TimeBlockRule {
    var id: UUID
    var name: String
    var startTime: Date // Time of day (only hour/minute matter)
    var endTime: Date // Time of day
    var linkedSetID: UUID? // Optional: which activity set to activate
    var isEnabled: Bool
    var daysOfWeek: [Int] // 1=Sunday, 2=Monday, etc. Empty = all days
    var createdAt: Date
    
    var trackingModeRaw: String
    
    var trackingMode: TrackingMode {
        get { TrackingMode(rawValue: trackingModeRaw) ?? .on }
        set { trackingModeRaw = newValue.rawValue }
    }
    
    init(name: String, startTime: Date, endTime: Date, linkedSetID: UUID? = nil, daysOfWeek: [Int] = [], trackingMode: TrackingMode = .on) {
        self.id = UUID()
        self.name = name
        self.startTime = startTime
        self.endTime = endTime
        self.linkedSetID = linkedSetID
        self.isEnabled = true
        self.daysOfWeek = daysOfWeek
        self.createdAt = Date()
        self.trackingModeRaw = trackingMode.rawValue
    }
    
    // Check if this rule applies to the current time
    func isActiveNow() -> Bool {
        guard isEnabled else { return false }
        
        let now = Date()
        let calendar = Calendar.current
        
        // Check day of week
        if !daysOfWeek.isEmpty {
            let currentWeekday = calendar.component(.weekday, from: now)
            guard daysOfWeek.contains(currentWeekday) else { return false }
        }
        
        // Check time range
        let nowComponents = calendar.dateComponents([.hour, .minute], from: now)
        let startComponents = calendar.dateComponents([.hour, .minute], from: startTime)
        let endComponents = calendar.dateComponents([.hour, .minute], from: endTime)
        
        guard let nowMinutes = nowComponents.hour.map({ $0 * 60 + (nowComponents.minute ?? 0) }),
              let startMinutes = startComponents.hour.map({ $0 * 60 + (startComponents.minute ?? 0) }),
              let endMinutes = endComponents.hour.map({ $0 * 60 + (endComponents.minute ?? 0) }) else {
            return false
        }
        
        // Handle overnight spans (e.g., 10pm - 2am)
        if endMinutes < startMinutes {
            return nowMinutes >= startMinutes || nowMinutes < endMinutes
        } else {
            return nowMinutes >= startMinutes && nowMinutes < endMinutes
        }
    }
    
    var timeRangeDisplay: String {
        let formatter = DateFormatter()
        formatter.timeStyle = .short
        return "\(formatter.string(from: startTime)) - \(formatter.string(from: endTime))"
    }
    
    var daysOfWeekDisplay: String {
        if daysOfWeek.isEmpty {
            return "Every day"
        }
        
        let dayNames = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"]
        return daysOfWeek.sorted().map { dayNames[$0 - 1] }.joined(separator: ", ")
    }
}

// MARK: - Time Block Model
@Model
final class TimeBlock {
    var id: UUID
    var activityID: UUID?  // nil if Gap or Pause
    var activityName: String
    var activityColor: String
    var activitySetID: UUID?
    
    var startTime: Date  // IMMUTABLE in UI
    var endTime: Date?
    
    var isPaused: Bool
    var pauseStartTime: Date?
    var totalPauseDuration: TimeInterval
    
    var stopState: String  // "switched", "stopped", "paused", "ongoing"
    var switchCount: Int
    var remarks: String
    var blockType: String  // "activity", "gap", "pause"
    
    init(activityID: UUID?, activityName: String, activityColor: String, activitySetID: UUID?, startTime: Date, blockType: String = "activity") {
        self.id = UUID()
        self.activityID = activityID
        self.activityName = activityName
        self.activityColor = activityColor
        self.activitySetID = activitySetID
        self.startTime = startTime
        self.endTime = nil
        self.isPaused = false
        self.pauseStartTime = nil
        self.totalPauseDuration = 0
        self.stopState = "ongoing"
        self.switchCount = 0
        self.remarks = ""
        self.blockType = blockType
    }
    
    convenience init(activity: Activity, startTime: Date) {
        self.init(
            activityID: activity.id,
            activityName: activity.name,
            activityColor: activity.color,
            activitySetID: activity.setID,
            startTime: startTime,
            blockType: "activity"
        )
    }
    
    static func createGap(startTime: Date, endTime: Date? = nil) -> TimeBlock {
        let gap = TimeBlock(
            activityID: nil,
            activityName: "Gap",
            activityColor: "#9E9E9E",
            activitySetID: nil,
            startTime: startTime,
            blockType: "gap"
        )
        gap.endTime = endTime
        return gap
    }
    
    static func createPause(startTime: Date, endTime: Date? = nil) -> TimeBlock {
        let pause = TimeBlock(
            activityID: nil,
            activityName: "Pause",
            activityColor: "#FF9800",
            activitySetID: nil,
            startTime: startTime,
            blockType: "pause"
        )
        pause.endTime = endTime
        return pause
    }
    
    var duration: TimeInterval {
        guard let end = endTime else {
            let currentDuration = Date().timeIntervalSince(startTime)
            return currentDuration - totalPauseDuration - (isPaused ? Date().timeIntervalSince(pauseStartTime ?? Date()) : 0)
        }
        return end.timeIntervalSince(startTime) - totalPauseDuration
    }
    
    var isActive: Bool { endTime == nil && !isPaused }
    var isGap: Bool { blockType == "gap" }
    var isPauseBlock: Bool { blockType == "pause" }
    var isRegularActivity: Bool { blockType == "activity" }
    
    var displayName: String {
        if isPauseBlock && switchCount > 0 {
            return "Pause (\(switchCount)x)"
        }
        return activityName
    }
}

// MARK: - Helper Extensions
extension TimeInterval {
    var formattedDuration: String {
        let hours = Int(self) / 3600
        let minutes = Int(self) / 60 % 60
        let seconds = Int(self) % 60
        
        if hours > 0 {
            return String(format: "%dh %dm", hours, minutes)
        } else if minutes > 0 {
            return String(format: "%dm %ds", minutes, seconds)
        } else {
            return String(format: "%ds", seconds)
        }
    }
    
    var detailedFormattedDuration: String {
        let hours = Int(self) / 3600
        let minutes = Int(self) / 60 % 60
        let seconds = Int(self) % 60
        
        if hours > 0 {
            return String(format: "%dh %dm %ds", hours, minutes, seconds)
        } else if minutes > 0 {
            return String(format: "%dm %ds", minutes, seconds)
        } else {
            return String(format: "%ds", seconds)
        }
    }
}

                        } else {

                        }
                    }
            } header: {
                Text("Notifications")
            }
            
            if notificationsEnabled {
                Section {
                    HStack {
                        Image(systemName: "bell.badge")
                            .foregroundStyle(.orange)
                        Text("You'll receive reminders before activities stop in background")
                            .font(.caption)
                    }
                    
                    HStack {
                        Image(systemName: "arrow.triangle.swap")
                            .foregroundStyle(.blue)
                        Text("You'll be notified when activity sets change automatically")
                            .font(.caption)
                    }
                } header: {
                    Text("Active Notifications")
                }
            }
        }
        .navigationTitle("Notifications")
        .navigationBarTitleDisplayMode(.inline)
        .onAppear {
            timeTrackingManager = TimeTrackingManager(modelContext: modelContext)
            notificationsEnabled = timeTrackingManager?.notificationsEnabled ?? true
            checkNotificationPermission()
        }
        .alert("Notifications Disabled", isPresented: $showingPermissionAlert) {
            Button("Open Settings") {
                if let url = URL(string: UIApplication.openSettingsURLString) {
                    UIApplication.shared.open(url)
                }
            }
            Button("Cancel", role: .cancel) {
                notificationsEnabled = false
                timeTrackingManager?.notificationsEnabled = false
            }
        } message: {
            Text("Please enable notifications in Settings to receive activity reminders")
        }
    }
    
    private func checkAndRequestNotificationPermission() {
        UNUserNotificationCenter.current().getNotificationSettings { settings in
            DispatchQueue.main.async {
                switch settings.authorizationStatus {
                case .authorized:
                    notificationsEnabled = true
                    timeTrackingManager?.notificationsEnabled = true
                case .notDetermined:
                    requestNotificationPermission()
                case .denied:
                    notificationsEnabled = false
                    timeTrackingManager?.notificationsEnabled = false
                    showingPermissionAlert = true
                case .provisional, .ephemeral:
                    notificationsEnabled = true
                    timeTrackingManager?.notificationsEnabled = true
                @unknown default:
                    notificationsEnabled = false
                    timeTrackingManager?.notificationsEnabled = false
                }
            }
        }
    }
    
    private func requestNotificationPermission() {
        UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .sound, .badge]) { granted, error in
            DispatchQueue.main.async {
                if granted {
                    notificationsEnabled = true
                    timeTrackingManager?.notificationsEnabled = true
                } else {
                    notificationsEnabled = false
                    timeTrackingManager?.notificationsEnabled = false
                    showingPermissionAlert = true
                }
            }
        }
    }
    
    private func checkNotificationPermission() {
        UNUserNotificationCenter.current().getNotificationSettings { settings in
            DispatchQueue.main.async {
                if settings.authorizationStatus == .denied {
                    notificationsEnabled = false
                    timeTrackingManager?.notificationsEnabled = false
                }
            }
        }
    }
}

// MARK: - Haptic Manager
class HapticManager {
    static let shared = HapticManager()
    
    private let impactLight = UIImpactFeedbackGenerator(style: .light)
    private let impactMedium = UIImpactFeedbackGenerator(style: .medium)
    private let impactHeavy = UIImpactFeedbackGenerator(style: .heavy)
    private let selectionGenerator = UISelectionFeedbackGenerator()
    private let notificationGenerator = UINotificationFeedbackGenerator()
    
    private init() {
        impactLight.prepare()
        impactMedium.prepare()
        impactHeavy.prepare()
        selectionGenerator.prepare()
        notificationGenerator.prepare()
    }
    
    func impact(style: UIImpactFeedbackGenerator.FeedbackStyle = .medium, intensity: CGFloat = 1.0) {
        guard UserDefaults.standard.bool(forKey: "hapticsEnabled") else { return }
        
        switch style {
        case .light:
            impactLight.impactOccurred(intensity: intensity)
        case .medium:
            impactMedium.impactOccurred(intensity: intensity)
        case .heavy:
            impactHeavy.impactOccurred(intensity: intensity)
        case .soft:
            impactLight.impactOccurred(intensity: intensity * 0.7)
        case .rigid:
            impactHeavy.impactOccurred(intensity: intensity * 1.2)
        @unknown default:
            impactMedium.impactOccurred(intensity: intensity)
        }
    }
    
    func selection() {
        guard UserDefaults.standard.bool(forKey: "hapticsEnabled") else { return }
        selectionGenerator.selectionChanged()
    }
    
    func notification(type: UINotificationFeedbackGenerator.FeedbackType) {
        guard UserDefaults.standard.bool(forKey: "hapticsEnabled") else { return }
        notificationGenerator.notificationOccurred(type)
    }
}

// MARK: - Date Section Enum

enum DateSection: String, CaseIterable, Identifiable {
    case today = "Today"
    case yesterday = "Yesterday"
    case twoDaysAgo = "2 Days Ago"
    case threeDaysAgo = "3 Days Ago"
    case fourDaysAgo = "4 Days Ago"
    case fiveDaysAgo = "5 Days Ago"
    case sixDaysAgo = "6 Days Ago"
    case lastWeek = "Last Week"
    case lastMonth = "Last Month"
    case lastThreeMonths = "Last 3 Months"
    case lastSixMonths = "Last 6 Months"
    case lastYear = "Last Year"
    case older = "Older"
    
    var id: String { rawValue }
    
    static func from(date: Date) -> DateSection {
        let calendar = Calendar.current
        let now = Date()
        
        if calendar.isDateInToday(date) { return .today }
        if calendar.isDateInYesterday(date) { return .yesterday }
        
        let daysAgo = calendar.dateComponents([.day], from: calendar.startOfDay(for: date), to: calendar.startOfDay(for: now)).day ?? 0
        
        switch daysAgo {
        case 2: return .twoDaysAgo
        case 3: return .threeDaysAgo
        case 4: return .fourDaysAgo
        case 5: return .fiveDaysAgo
        case 6: return .sixDaysAgo
        case 7...30: return .lastWeek
        case 31...90: return .lastMonth
        case 91...180: return .lastThreeMonths
        case 181...365: return .lastSixMonths
        case 366...730: return .lastYear
        default: return .older
        }
    }
    
    var icon: String {
        switch self {
        case .today: return "sun.max.fill"
        case .yesterday: return "moon.fill"
        case .twoDaysAgo, .threeDaysAgo, .fourDaysAgo, .fiveDaysAgo, .sixDaysAgo, .lastWeek: return "calendar"
        case .lastMonth: return "calendar.badge.clock"
        case .lastThreeMonths, .lastSixMonths: return "calendar.circle"
        case .lastYear, .older: return "archivebox"
        }
    }
}

struct ReadOnlyTimelineView: View {

            .reduce(0) { $0 + $1.duration }
    }
    
    private var gaps: [TimeBlock] {
        blocks.filter { $0.isGap }
    }

        gaps.reduce(0) { $0 + $1.duration }
    }

                        .font(.caption)
                        .foregroundStyle(.secondary)
                }
                Spacer()
            }
            .padding()
            .background(Color.orange.opacity(0.1))
            
            // Summary (same as AuditListView)
            DaySummaryBar(
                date: date,
                totalTracked: totalTracked,
                gapCount: gaps.count,
                totalGapTime: totalGapTime,
                onResolveAllGaps: { } // No-op in read-only mode
            )
            
            if blocks.isEmpty {
                ContentUnavailableView(
                    "No Activities",
                    systemImage: "clock",
                    description: Text("No time blocks recorded for this day")
                )
            } else {
                List {
                    ForEach(sortedBlocks) { block in
                        ReadOnlyBlockRow(block: block)
                            .listRowInsets(EdgeInsets(top: 4, leading: 16, bottom: 4, trailing: 16))
                    }
                }
                .listStyle(.plain)
            }
        }
        .navigationTitle(date.formatted(date: .abbreviated, time: .omitted))
        .onAppear {
            loadBlocks()
        }
    }
    
    private func loadBlocks() {
        let calendar = Calendar.current
        let dayStart = calendar.startOfDay(for: date)
        let dayEnd = calendar.date(byAdding: .day, value: 1, to: dayStart)!
        
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { block in
                block.startTime >= dayStart && block.startTime < dayEnd
            },
            sortBy: [SortDescriptor(\.startTime)]
        )
        
        blocks = (try? modelContext.fetch(descriptor)) ?? []
    }

import SwiftUI

// MARK: - Color Hex Extension
extension Color {
    init(hex: String) {
        let hex = hex.trimmingCharacters(in: CharacterSet.alphanumerics.inverted)
        var int: UInt64 = 0
        Scanner(string: hex).scanHexInt64(&int)
        let a, r, g, b: UInt64
        switch hex.count {
        case 3:
            (a, r, g, b) = (255, (int >> 8) * 17, (int >> 4 & 0xF) * 17, (int & 0xF) * 17)
        case 6:
            (a, r, g, b) = (255, int >> 16, int >> 8 & 0xFF, int & 0xFF)
        case 8:
            (a, r, g, b) = (int >> 24, int >> 16 & 0xFF, int >> 8 & 0xFF, int & 0xFF)
        default:
            (a, r, g, b) = (255, 0, 0, 0)
        }
        
        self.init(
            .sRGB,
            red: Double(r) / 255,
            green: Double(g) / 255,
            blue: Double(b) / 255,
            opacity: Double(a) / 255
        )
    }
}

import SwiftUI
import SwiftData

struct AddBlockSheet: View {
    @Environment(\.modelContext) private var modelContext
    @Environment(\.dismiss) private var dismiss
    
    let date: Date
    let conflictDetector: ConflictDetector
    let onSave: () -> Void
    
    @Query private var activities: [Activity]
    
    @State private var selectedActivity: Activity?
    @State private var startTime: Date
    @State private var endTime: Date
    @State private var remarks = ""
    @State private var validationResult: ValidationResult?
    @State private var showingConflictAlert = false
    
    init(date: Date, conflictDetector: ConflictDetector, onSave: @escaping () -> Void) {
        self.date = date
        self.conflictDetector = conflictDetector
        self.onSave = onSave
        
        // Default to current hour
        let calendar = Calendar.current
        let now = Date()
        let components = calendar.dateComponents([.year, .month, .day, .hour], from: date)
        let startOfHour = calendar.date(from: components) ?? now
        
        _startTime = State(initialValue: startOfHour)
        _endTime = State(initialValue: startOfHour.addingTimeInterval(3600))
    }
    
    private var duration: TimeInterval {
        endTime.timeIntervalSince(startTime)
    }
    
    private var canSave: Bool {
        selectedActivity != nil &&
        duration > 0 &&
        (validationResult?.canProceedWithConfirmation ?? false)
    }
    
    var body: some View {
        NavigationStack {
            Form {
                // Activity selection
                Section("Activity") {
                    if let activity = selectedActivity {
                        HStack {
                            if let emoji = activity.emoji {
                                Text(emoji)
                            }
                            Circle()
                                .fill(Color(activity.color))
                                .frame(width: 20, height: 20)
                            Text(activity.name)
                            Spacer()
                            Button("Change") {
                                selectedActivity = nil
                            }
                            .font(.caption)
                        }
                    } else {
                        ForEach(activities.prefix(8)) { activity in
                            Button {
                                selectedActivity = activity
                            } label: {
                                HStack {
                                    if let emoji = activity.emoji {
                                        Text(emoji)
                                    }
                                    Circle()
                                        .fill(Color(activity.color))
                                        .frame(width: 16, height: 16)
                                    Text(activity.name)
                                        .foregroundStyle(.primary)
                                }
                            }
                        }
                    }
                }
                
                // Time selection
                Section("Time") {
                    DatePicker("Start", selection: $startTime, displayedComponents: .hourAndMinute)
                    DatePicker("End", selection: $endTime, displayedComponents: .hourAndMinute)
                    
                    HStack {
                        Text("Duration")
                        Spacer()
                        Text(duration.formattedDuration)
                            .foregroundStyle(duration > 0 ? Color.primary : Color.red)
                            .monospacedDigit()
                    }
                }
                
                // Conflicts
                if let validation = validationResult, !validation.isValid {
                    Section {
                        ForEach(validation.conflicts) { conflict in
                            HStack {
                                Image(systemName: "exclamationmark.triangle.fill")
                                    .foregroundStyle(conflict.severity == .blocking ? .red : .orange)
                                Text(conflict.description)
                                    .font(.subheadline)
                            }
                        }
                    }
                }
                
                // Notes
                Section("Notes (Optional)") {
                    TextField("What were you doing?", text: $remarks, axis: .vertical)
                        .lineLimit(3...6)
                }
            }
            .navigationTitle("Add Block")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .cancellationAction) {
                    Button("Cancel") { dismiss() }
                }
                
                ToolbarItem(placement: .confirmationAction) {
                    Button("Add") {
                        addBlock()
                    }
                    .fontWeight(.semibold)
                    .disabled(!canSave)
                }
            }
            .onChange(of: startTime) { _, _ in validate() }
            .onChange(of: endTime) { _, _ in validate() }
            .onAppear { validate() }
        }
        .presentationDetents([.large])
    }
    
    private func validate() {
        // Create a temporary block for validation
        let tempBlock = TimeBlock(
            activityID: selectedActivity?.id,
            activityName: selectedActivity?.name ?? "",
            activityColor: selectedActivity?.color ?? "#999999",
            activitySetID: selectedActivity?.setID,
            startTime: startTime
        )
        tempBlock.endTime = endTime
        
        validationResult = conflictDetector.validateEdit(
            block: tempBlock,
            newStart: startTime,
            newEnd: endTime
        )
    }
    
    private func addBlock() {
        guard let activity = selectedActivity else { return }
        
        let newBlock = TimeBlock(activity: activity, startTime: startTime)
        newBlock.endTime = endTime
        newBlock.remarks = remarks
        newBlock.stopState = "stopped"
        
        modelContext.insert(newBlock)
        try? modelContext.save()
        
        onSave()
        dismiss()
    }
}

    private func loadRecentActivities() {
        // Find activities used in the last 7 days
        let weekAgo = Calendar.current.date(byAdding: .day, value: -7, to: Date())!
        
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { block in
                block.startTime >= weekAgo && block.blockType == "activity"
            },
            sortBy: [SortDescriptor(\.startTime, order: .reverse)]
        )
        
        guard let recentBlocks = try? modelContext.fetch(descriptor) else { return }
        
        // Get unique activity IDs in order of recency
        var seen = Set<UUID>()
        var recentIDs: [UUID] = []
        
        for block in recentBlocks {
            if let id = block.activityID, !seen.contains(id) {
                seen.insert(id)
                recentIDs.append(id)
            }
        }
        
        // Map to activities
        recentActivities = recentIDs.compactMap { id in
            activities.first { $0.id == id }
        }
    }
    
    private func resolveGap(with activity: Activity) {
        // Convert the gap to an activity
        gap.activityID = activity.id
        gap.activityName = activity.name
        gap.activityColor = activity.color
        gap.activitySetID = activity.setID
        gap.blockType = "activity"
        
        // Apply time adjustments if changed
        if adjustedStartTime != gap.startTime {
            gap.startTime = adjustedStartTime
        }
        if adjustedEndTime != (gap.endTime ?? Date()) {
            gap.endTime = adjustedEndTime
        }
        
        try? modelContext.save()
        onResolved()
        dismiss()
    }

        _editedRemarks = State(initialValue: block.remarks)
    }

        editedStartTime != block.startTime ||
        editedEndTime != (block.endTime ?? Date()) ||
        editedActivityID != block.activityID ||

                    timeEditors

        // SAVE OLD VALUES BEFORE CHANGING
        let oldStart = block.startTime
        let oldEnd = block.endTime
        
        // Update the block with edited values
        block.startTime = editedStartTime
        if block.endTime != nil {
            block.endTime = editedEndTime

        // Update activity if changed
        if let newActivityID = editedActivityID,
           newActivityID != block.activityID,
           let activity = activities.first(where: { $0.id == newActivityID }) {
            block.activityID = activity.id
            block.activityName = activity.name
            block.activityColor = activity.color
            block.activitySetID = activity.setID
            block.blockType = "activity"
        }
        
        block.remarks = editedRemarks
        
        // CALL AUTO-FILL LOGIC
        handleTimestampAdjustments(
            oldStart: oldStart,
            oldEnd: oldEnd,
            newStart: editedStartTime,
            newEnd: editedEndTime
        )
        
        try? modelContext.save()
        onSave()
        dismiss()
    }
    
    // MARK: - Auto-Fill Logic

    /// Handles creation of gaps/pauses when timestamp edits create empty space
    private func handleTimestampAdjustments(
        oldStart: Date,
        oldEnd: Date?,
        newStart: Date,
        newEnd: Date
    ) {
        let (previousActivity, nextActivity) = getAdjacentActivityBlocks(for: block)
        
        // Handle space created BEFORE this block (moved start time forward)
        if newStart > oldStart {
            let spaceStart = previousActivity?.endTime ?? oldStart
            handleSpace(from: spaceStart, to: newStart, extendingBlock: previousActivity)
        }
        
        // Handle space created AFTER this block (moved end time backward)
        if let oldEnd = oldEnd, newEnd < oldEnd {
            let spaceEnd = nextActivity?.startTime ?? oldEnd
            handleSpace(from: newEnd, to: spaceEnd, extendingBlock: block)
        }
        
        // Clean up any absorbed gaps/pauses
        cleanupAbsorbedBlocks(newStart: newStart, newEnd: newEnd)
    }

    /// Creates gap/pause or auto-snaps based on duration
    private func handleSpace(from start: Date, to end: Date, extendingBlock: TimeBlock?) {
        let duration = end.timeIntervalSince(start)
        
        if duration < 60 {
            // < 1 minute: Auto-snap by extending adjacent block
            if let blockToExtend = extendingBlock {
                blockToExtend.endTime = end
            } else {
                // If no block to extend, extend current block backward
                block.startTime = start
            }
        } else if duration < 300 {
            // 1-5 minutes: Create Pause
            let pause = TimeBlock.createPause(startTime: start, endTime: end)
            pause.stopState = "stopped"
            modelContext.insert(pause)
        } else {
            // ≥ 5 minutes: Create Gap
            let gap = TimeBlock.createGap(startTime: start, endTime: end)
            gap.stopState = "stopped"
            modelContext.insert(gap)
        }
    }

    /// Removes gaps/pauses that have been absorbed by timestamp expansion
    private func cleanupAbsorbedBlocks(newStart: Date, newEnd: Date) {
        let calendar = Calendar.current
        let dayStart = calendar.startOfDay(for: block.startTime)
        let dayEnd = calendar.date(byAdding: .day, value: 1, to: dayStart)!
        
        // Simplified predicate - fetch all gaps/pauses for the day
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { b in
                b.startTime >= dayStart &&
                b.startTime < dayEnd
            }
        )
        
        guard let allBlocks = try? modelContext.fetch(descriptor) else { return }
        
        // Filter in code instead of in predicate (easier for compiler)
        let gapsAndPauses = allBlocks.filter { candidate in
            candidate.id != block.id &&
            (candidate.blockType == "gap" || candidate.blockType == "pause")
        }
        
        for candidate in gapsAndPauses {
            guard let candidateEnd = candidate.endTime else { continue }
            
            // Fully absorbed (completely inside the edited block)
            if candidate.startTime >= newStart && candidateEnd <= newEnd {
                modelContext.delete(candidate)
            }
            // Partially absorbed from the end (trim it)
            else if candidate.startTime < newStart && candidateEnd > newStart && candidateEnd <= newEnd {
                candidate.endTime = newStart
            }
            // Partially absorbed from the start (trim it)
            else if candidate.startTime >= newStart && candidate.startTime < newEnd && candidateEnd > newEnd {
                candidate.startTime = newEnd
                
                // Re-check duration after trimming - delete if too small
                let newDuration = candidateEnd.timeIntervalSince(newEnd)
                if newDuration < 60 {
                    modelContext.delete(candidate)
                }
            }
        }
    }

    /// Gets adjacent ACTIVITY blocks (skipping gaps/pauses)
    private func getAdjacentActivityBlocks(for block: TimeBlock) -> (previous: TimeBlock?, next: TimeBlock?) {
        let calendar = Calendar.current
        let dayStart = calendar.startOfDay(for: block.startTime)
        let dayEnd = calendar.date(byAdding: .day, value: 1, to: dayStart)!
        
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { b in
                b.startTime >= dayStart &&
                b.startTime < dayEnd &&
                b.blockType == "activity"  // Only actual activities
            },
            sortBy: [SortDescriptor(\.startTime, order: .forward)]
        )
        
        guard let allActivities = try? modelContext.fetch(descriptor),
              let currentIndex = allActivities.firstIndex(where: { $0.id == block.id }) else {
            return (nil, nil)
        }
        
        // Get previous activity (if any)
        let previous = currentIndex > 0 ? allActivities[currentIndex - 1] : nil
        
        // Get next activity (if any)
        let next = currentIndex < allActivities.count - 1 ? allActivities[currentIndex + 1] : nil
        
        return (previous, next)
    }
}

import SwiftUI
import SwiftData

struct SplitBlockSheet: View {
    @Environment(\.modelContext) private var modelContext
    @Environment(\.dismiss) private var dismiss
    
    let block: TimeBlock
    let onSplit: () -> Void
    
    @State private var splitTime: Date
    @State private var showingError = false
    @State private var errorMessage = ""
    
    init(block: TimeBlock, onSplit: @escaping () -> Void) {
        self.block = block
        self.onSplit = onSplit
        
        // Default to midpoint
        let midpoint = block.startTime.addingTimeInterval(block.duration / 2)
        _splitTime = State(initialValue: midpoint)
    }
    
    private var firstPartDuration: TimeInterval {
        splitTime.timeIntervalSince(block.startTime)
    }
    
    private var secondPartDuration: TimeInterval {
        (block.endTime ?? Date()).timeIntervalSince(splitTime)
    }
    
    private var isValidSplit: Bool {
        firstPartDuration >= 60 && secondPartDuration >= 60 // Minimum 1 minute each
    }
    
    var body: some View {
        NavigationStack {
            Form {
                // Original block info
                Section {
                    HStack {
                        Circle()
                            .fill(Color(block.activityColor))
                            .frame(width: 24, height: 24)
                        
                        VStack(alignment: .leading) {
                            Text(block.displayName)
                                .font(.headline)
                            Text("\(block.startTime.formatted(date: .omitted, time: .shortened)) → \((block.endTime ?? Date()).formatted(date: .omitted, time: .shortened))")
                                .font(.caption)
                                .foregroundStyle(.secondary)
                        }
                        
                        Spacer()
                        
                        Text(block.duration.formattedDuration)
                            .font(.headline)
                            .monospacedDigit()
                    }
                } header: {
                    Text("Original Block")
                }
                
                // Split time picker
                Section {
                    DatePicker(
                        "Split At",
                        selection: $splitTime,
                        in: block.startTime...(block.endTime ?? Date()),
                        displayedComponents: .hourAndMinute
                    )
                    
                    // Quick presets
                    HStack {
                        Text("Quick:")
                            .font(.caption)
                            .foregroundStyle(.secondary)
                        
                        Button("¼") {
                            splitTime = block.startTime.addingTimeInterval(block.duration * 0.25)
                        }
                        .buttonStyle(.bordered)
                        
                        Button("½") {
                            splitTime = block.startTime.addingTimeInterval(block.duration * 0.5)
                        }
                        .buttonStyle(.bordered)
                        
                        Button("¾") {
                            splitTime = block.startTime.addingTimeInterval(block.duration * 0.75)
                        }
                        .buttonStyle(.bordered)
                    }
                } header: {
                    Text("Split Point")
                }
                
                // Preview of result
                Section {
                    HStack {
                        Text("Part 1")
                        Spacer()
                        Text("\(block.startTime.formatted(date: .omitted, time: .shortened)) → \(splitTime.formatted(date: .omitted, time: .shortened))")
                            .foregroundStyle(.secondary)
                        Text(firstPartDuration.formattedDuration)
                            .fontWeight(.medium)
                            .monospacedDigit()
                    }
                    
                    HStack {
                        Text("Part 2")
                        Spacer()
                        Text("\(splitTime.formatted(date: .omitted, time: .shortened)) → \((block.endTime ?? Date()).formatted(date: .omitted, time: .shortened))")
                            .foregroundStyle(.secondary)
                        Text(secondPartDuration.formattedDuration)
                            .fontWeight(.medium)
                            .monospacedDigit()
                    }
                } header: {
                    Text("Result Preview")
                } footer: {
                    Text("Both parts will keep the same activity. You can change this after splitting.")
                }
            }
            .navigationTitle("Split Block")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .cancellationAction) {
                    Button("Cancel") { dismiss() }
                }
                
                ToolbarItem(placement: .confirmationAction) {
                    Button("Split") {
                        performSplit()
                    }
                    .fontWeight(.semibold)
                    .disabled(!isValidSplit)
                }
            }
            .alert("Cannot Split", isPresented: $showingError) {
                Button("OK") { }
            } message: {
                Text(errorMessage)
            }
        }
        .presentationDetents([.medium])
    }
    
    private func performSplit() {
        let editor = TimeBlockEditor(modelContext: modelContext)
        
        do {
            try editor.splitBlock(block, at: splitTime)
            onSplit()
            dismiss()
        } catch {
            errorMessage = error.localizedDescription
            showingError = true
        }
    }
}

    @State private var gapToResolve: TimeBlock?
    @State private var conflictDetector: ConflictDetector?

    @State private var isSelectionMode = false
    @State private var selectedBlockIDs: Set<UUID> = []
    
    private var sortedBlocks: [TimeBlock] {
        blocks.sorted { $0.startTime < $1.startTime }
    }
    
    private var gaps: [TimeBlock] {
        blocks.filter { $0.isGap }
    }
    
    private var totalTracked: TimeInterval {
        blocks.filter { !$0.isGap && !$0.isPauseBlock }
            .reduce(0) { $0 + $1.duration }
    }
    
    private var totalGapTime: TimeInterval {
        gaps.reduce(0) { $0 + $1.duration }
    }
    
    var body: some View {
        VStack(spacing: 0) {
            // Summary Header
            DaySummaryBar(
                date: date,
                totalTracked: totalTracked,
                gapCount: gaps.count,
                totalGapTime: totalGapTime,
                onResolveAllGaps: { showBulkGapResolver() }
            )
            
            if blocks.isEmpty {
                emptyState
            } else {
                blockList
            }
            
            bottomToolbar
        }
        .navigationTitle(date.formatted(date: .abbreviated, time: .omitted))
        .navigationBarTitleDisplayMode(.inline)
        .onAppear {
            conflictDetector = ConflictDetector(modelContext: modelContext)
            loadBlocks()
        }
        .refreshable {
            loadBlocks()
        }
        .sheet(item: $selectedBlock) { block in
            BlockEditSheet(
                block: block,
                conflictDetector: conflictDetector!,
                onSave: { loadBlocks() }

                    }

                Button {

    private func loadBlocks() {
        let calendar = Calendar.current

        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { block in
                block.startTime >= dayStart && block.startTime < dayEnd
            },
            sortBy: [SortDescriptor(\.startTime)]
        )
        
        blocks = (try? modelContext.fetch(descriptor)) ?? []
    }
    
    private func toggleSelection(_ block: TimeBlock) {
        if selectedBlockIDs.contains(block.id) {
            selectedBlockIDs.remove(block.id)
        } else {
            selectedBlockIDs.insert(block.id)
        }
    }
    
    private func mergeSelected() {
        // Get blocks in order
        let toMerge = sortedBlocks.filter { selectedBlockIDs.contains($0.id) }
        guard toMerge.count >= 2 else { return }
        
        // Check if they're adjacent
        let editor = TimeBlockEditor(modelContext: modelContext)
        guard editor.canMerge(toMerge) else {
            // Show error
            return
        }
        
        // Use the first block's activity for the merged result
        let targetActivity = findActivity(for: toMerge.first!)
        
        do {
            try editor.mergeBlocks(toMerge, intoActivity: targetActivity)
            selectedBlockIDs.removeAll()
            isSelectionMode = false
            loadBlocks()
        } catch {
            print("Merge failed: \(error)")
        }
    }
    
    private func showBulkGapResolver() {
        // TODO: Implement bulk gap resolution
    }
    
    private func findActivity(for block: TimeBlock) -> Activity? {
        guard let activityID = block.activityID else { return nil }
        let descriptor = FetchDescriptor<Activity>(
            predicate: #Predicate { $0.id == activityID }
        )
        return try? modelContext.fetch(descriptor).first
    }
}

struct TrackingView: View {
    @ObservedObject var manager: TimeTrackingManager
    @Environment(\.modelContext) private var modelContext
    
    @Query private var activitySets: [ActivitySet]
    @Query private var activities: [Activity]
    
    @State private var timer = Timer.publish(every: 1, on: .main, in: .common).autoconnect()
    @State private var showingQuickRemark = false
    
    var activeSet: ActivitySet? {
        activitySets.first { $0.isActive }
    }
    
    var activeSetActivities: [Activity] {
        guard let setID = activeSet?.id else { return [] }
        return activities.filter { $0.setID == setID }.sorted { $0.sortOrder < $1.sortOrder }
    }
    
    // Check if current block is a Pause
    var isCurrentlyPaused: Bool {
        manager.currentTimeBlock?.isPauseBlock ?? false
    }
    
    // Get the last activity before pause (for Resume button)
    var lastActivityBeforePause: Activity? {
        guard isCurrentlyPaused else { return nil }
        
        // Find the most recent non-pause/non-gap activity
        let descriptor = FetchDescriptor<TimeBlock>(
            predicate: #Predicate { block in
                block.blockType == "activity" && block.endTime != nil
            },
            sortBy: [SortDescriptor(\.endTime, order: .reverse)]
        )
        
        guard let lastActivityBlock = try? modelContext.fetch(descriptor).first,
              let activityID = lastActivityBlock.activityID,
              let activity = activities.first(where: { $0.id == activityID }) else {
            return nil
        }
        
        return activity
    }
    
    var body: some View {
        ScrollView {
            VStack(spacing: 20) {
                // Current Activity Display
                CurrentActivityCard(
                    block: manager.currentTimeBlock,
                    isPaused: isCurrentlyPaused,
                    onPause: {
                        manager.pauseCurrentActivity()
                    },
                    onStop: {
                        manager.stopCurrentActivity()
                    },
                    onResume: {
                        if let activity = lastActivityBeforePause {
                            manager.resumeFromPause(activity)
                        }
                    },
                    onQuickRemark: {
                        showingQuickRemark = true
                    }
                )
                
                // Activity Set Picker - moved below current activity
                if activitySets.count > 1 {
                    ActivitySetPicker(sets: activitySets)
                        .padding(.horizontal)
                }
                
                // Activity Buttons Grid
                LazyVGrid(columns: [
                    GridItem(.flexible()),
                    GridItem(.flexible()),
                    GridItem(.flexible())
                ], spacing: 12) {
                    ForEach(activeSetActivities) { activity in
                        ActivityButton(
                            activity: activity,
                            isActive: manager.currentTimeBlock?.activityID == activity.id && !isCurrentlyPaused
                        ) {
                            if isCurrentlyPaused {
                                // If currently paused, resume with this activity
                                manager.resumeFromPause(activity)
                            } else {
                                // Normal activity start
                                manager.startActivity(activity)
                            }
                        }
                    }
                }
                .padding(.horizontal)
            }
            .padding(.top)
        }
        .onReceive(timer) { _ in
            if manager.currentTimeBlock != nil {
                manager.objectWillChange.send()
            }
        }
        .sheet(isPresented: $showingQuickRemark) {
            if let block = manager.currentTimeBlock {
                QuickRemarkSheet(timeBlock: block)
            }
        }
    }
}

// MARK: - Quick Remark Sheet
struct QuickRemarkSheet: View {
    @Environment(\.modelContext) private var modelContext
    @Environment(\.dismiss) private var dismiss
    
    @Bindable var timeBlock: TimeBlock
    @State private var newRemark = ""
    @FocusState private var isTextFieldFocused: Bool
    
    var body: some View {
        NavigationStack {
            VStack(spacing: 20) {
                // Current activity info
                VStack(spacing: 8) {
                    HStack {
                        Circle()
                            .fill(Color(timeBlock.activityColor))
                            .frame(width: 12, height: 12)
                        
                        Text(timeBlock.displayName)
                            .font(.subheadline)
                            .foregroundStyle(.secondary)
                        
                        Spacer()
                        
                        Text(timeBlock.duration.formattedDuration)
                            .font(.caption)
                            .foregroundStyle(.secondary)
                            .monospacedDigit()
                    }
                    .padding()
                    .background(Color(.systemGray6))
                    .cornerRadius(12)
                }
                
                // Input field
                VStack(alignment: .leading, spacing: 8) {
                    Text("Add Quick Remark")
                        .font(.headline)
                    
                    TextField("Type your note...", text: $newRemark, axis: .vertical)
                        .textFieldStyle(.roundedBorder)
                        .focused($isTextFieldFocused)
                        .lineLimit(3...6)
                        .onSubmit {
                            // Submit on Enter/Return key
                            if !newRemark.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
                                addRemark()
                            }
                        }
                        .submitLabel(.done)
                }
                
                // Existing remarks preview
                if !timeBlock.remarks.isEmpty {
                    VStack(alignment: .leading, spacing: 8) {
                        Text("Previous Remarks")
                            .font(.caption)
                            .foregroundStyle(.secondary)
                        
                        ScrollView {
                            Text(timeBlock.remarks)
                                .font(.caption)
                                .foregroundStyle(.secondary)
                                .frame(maxWidth: .infinity, alignment: .leading)
                                .padding()
                                .background(Color(.systemGray6))
                                .cornerRadius(8)
                        }
                        .frame(maxHeight: 120)
                    }
                }
                
                Spacer()
            }
            .padding()
            .navigationTitle("Quick Remark")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .cancellationAction) {
                    Button("Cancel") {
                        dismiss()
                    }
                }
                
                ToolbarItem(placement: .confirmationAction) {
                    Button("Add") {
                        addRemark()
                    }
                    .disabled(newRemark.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty)
                    .fontWeight(.semibold)
                }
                
                // Keyboard toolbar with Done button
                ToolbarItemGroup(placement: .keyboard) {
                    Spacer()
                    Button("Done") {
                        isTextFieldFocused = false
                    }
                }
            }
            .onAppear {
                // Auto-focus the text field
                DispatchQueue.main.asyncAfter(deadline: .now() + 0.5) {
                    isTextFieldFocused = true
                }
            }
        }
        .presentationDetents([.medium, .large])
        .presentationDragIndicator(.visible)
    }
    
    private func addRemark() {
        let trimmedRemark = newRemark.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmedRemark.isEmpty else { return }
        
        // Get current timestamp
        let timestamp = Date()
        let formatter = DateFormatter()
        formatter.dateFormat = "HH:mm"
        let timeString = formatter.string(from: timestamp)
        
        // Format the new remark with timestamp
        let formattedRemark = "[\(timeString)] \(trimmedRemark)"
        
        // Append to existing remarks with newline separator
        if timeBlock.remarks.isEmpty {
            timeBlock.remarks = formattedRemark
        } else {
            timeBlock.remarks += "\n" + formattedRemark
        }
        
        // Save context
        try? modelContext.save()
        
        dismiss()
    }
}

struct ReportsView: View {

    @State private var selectedPeriod: ReportGenerator.ReportPeriod = .thisWeek
    @State private var report: ReportGenerator.Report?
    @State private var showingShareSheet = false
    @State private var reportText = ""
    
    let reportGenerator = ReportGenerator()
    
    var body: some View {
        NavigationStack {
            ScrollView {
                VStack(spacing: 24) {
                    Picker("Period", selection: $selectedPeriod) {
                        Text("This Week").tag(ReportGenerator.ReportPeriod.thisWeek)
                        Text("Last Week").tag(ReportGenerator.ReportPeriod.lastWeek)
                        Text("This Month").tag(ReportGenerator.ReportPeriod.thisMonth)
                        Text("Last Month").tag(ReportGenerator.ReportPeriod.lastMonth)
                    }
                    .pickerStyle(.segmented)
                    .padding(.horizontal)
                    
                    if let report = report {
                        // Summaries
                        HStack(spacing: 12) {
                            SummaryCard(title: "Total Time", value: report.totalDuration.formattedDuration, icon: "clock.fill", color: .blue)
                            SummaryCard(title: "Activities", value: "\(report.totalBlocks)", icon: "list.bullet", color: .green)
                        }
                        .padding(.horizontal)
                        
                        // Comparison
                        if let comparison = report.comparisonToPrevious {
                            ComparisonCard(comparison: comparison)
                                .padding(.horizontal)
                        }
                        
                        // Top Activities
                        if !report.topActivities.isEmpty {
                            TopActivitiesCard(report: report)
                                .padding(.horizontal)
                        }
                        
                        // Charts
                        if !report.activityBreakdown.isEmpty {
                            BreakdownChart(report: report)
                                .padding(.horizontal)
                        }
                        
                    } else {
                        ContentUnavailableView("No Data", systemImage: "chart.bar.xaxis")
                    }
                }
                .padding(.vertical)
            }
            .navigationTitle("Reports")
            .toolbar {
                ToolbarItem(placement: .topBarTrailing) {
                    Button(action: shareReport) {
                        Image(systemName: "square.and.arrow.up")
                    }
                    .disabled(report == nil)
                }
            }
            .sheet(isPresented: $showingShareSheet) {
                if let report = report {
                    ShareSheet(text: reportGenerator.formatReportAsText(report))
                }
            }
            .onChange(of: selectedPeriod) { _, _ in generateReport() }
            .onAppear { generateReport() }
        }
    }
    
    private func generateReport() {
        report = reportGenerator.generateReport(for: selectedPeriod, blocks: allTimeBlocks, activities: activities)
    }
    
    private func shareReport() {
        guard let report = report else { return }
        reportText = reportGenerator.formatReportAsText(report)
        showingShareSheet = true
    }
}

// MARK: - The Isolated Row View (Fixes the logic errors)
struct ActivityChangeRow: View {
    let change: ActivityChange
    
    private var color: Color {
        let isIncrease = change.change > 0
        switch change.goal {
        case .increase:
            return isIncrease ? .green : .red
        case .decrease:
            return isIncrease ? .red : .green
        case .neutral:
            return .secondary
        }
    }
    private var icon: String {
        let isIncrease = change.change > 0
        switch change.goal {
        case .increase, .decrease:
            return isIncrease ? "arrow.up" : "arrow.down"
        case .neutral:
            return isIncrease ? "arrow.up" : "arrow.down"
        }
    }
    
    var body: some View {
        HStack {
            HStack(spacing: 6) {
                Image(systemName: icon)
                    .font(.caption)
                    .foregroundStyle(color)
                    .frame(width: 12)
                
                Text(change.name)
                    .font(.caption)
                    .fontWeight(.medium)
            }
            
            Spacer()
            
            Text(abs(change.change).formattedDuration)
                .font(.caption)
                .monospacedDigit()
                .foregroundStyle(color)
        }
    }
}

                                    percentage: stats.totalDuration > 0 ? (activity.duration / stats.totalDuration) * 100 : 0,
                                    color: getActivityColor(activity.name)
                                )
                            }
                        }
                        .padding(.vertical)
                        .background(Color(.systemGray6))
                        .cornerRadius(12)
                        .padding(.horizontal)
                        
                    } else {
                        ContentUnavailableView(
                            "No Data Yet",
                            systemImage: "chart.bar",
                            description: Text("Start tracking activities to see your stats")
                        )
                    }
                }
                .padding(.vertical)
            }
            .navigationTitle("Today's Stats")
            .toolbar {
                ToolbarItem(placement: .topBarTrailing) {
                    Button(action: loadStats) {
                        Image(systemName: "arrow.clockwise")
                    }
                }
            }
            .onAppear {
                loadStats()
                startAutoRefresh()
            }
        }
    }
    
    private func loadStats() {
        stats = manager.getTodaysStats()
        timeBlocks = manager.getTodaysTimeBlocks()
    }
    
    private func startAutoRefresh() {
        Timer.scheduledTimer(withTimeInterval: 10.0, repeats: true) { _ in
            loadStats()
        }
    }
    
    private func getActivityColor(_ name: String) -> Color {
        // Find the color from time blocks
        if let block = timeBlocks.first(where: { $0.activityName == name }) {
            return Color(block.activityColor)
        }
        return .gray
    }
}

struct TimelineChart: View {
    let blocks: [TimeBlock]
    
    private var startOfDay: Date {
        Calendar.current.startOfDay(for: Date())
    }
    
    private var endOfDay: Date {
        Calendar.current.date(byAdding: .day, value: 1, to: startOfDay)!
    }
    
    var body: some View {
        GeometryReader { geometry in
            ZStack(alignment: .leading) {
                // Background
                Rectangle()
                    .fill(Color(.systemGray5))
                
                // Time blocks
                ForEach(blocks.filter { $0.endTime != nil }) { block in
                    let start = block.startTime
                    let end = block.endTime ?? Date()
                    
                    let startOffset = start.timeIntervalSince(startOfDay) / 86400
                    let duration = end.timeIntervalSince(start) / 86400
                    
                    Rectangle()
                        .fill(Color(block.activityColor))
                        .frame(width: geometry.size.width * duration)
                        .offset(x: geometry.size.width * startOffset)
                }
            }
        }
        .cornerRadius(8)
    }
}
